- SQuAD 形式: `context`（文書）、`question`（質問）、`answers`（答えの文字列と開始位置のリスト）。評価データには 1 つの質問に複数の正解が付いていることがある。

- JSQuAD の質問はすべて答えが文書中にある（SQuAD 1.1 相当）。答えのない質問を含む設定はこの章の「文書に答えがない質問を「答えなし」と判定する」のレシピを参照。
- 最近の `datasets` は、読み込みスクリプトを使う古い形式のデータセットを読めない。Parquet 形式で公開されているリポジトリを選ぶ（ここでは `sbintuitions/JSQuAD`）。

## 文書の中から質問の答えを抜き出す（抽出型 QA） {extractive_qa}

文書と質問を与えて、文書中の答えに当たる箇所を抜き出したい。

```python
import sentencepiece as spm
import torch
from huggingface_hub import hf_hub_download
from tokenizers import models
from transformers import AutoModelForQuestionAnswering, AutoTokenizer

name = "tsmatz/roberta_qa_japanese"
tokenizer = AutoTokenizer.from_pretrained(name)
# transformers 5 系の T5Tokenizer は未知語 (大文字の英字など) を文末記号 (</s>) の ID にしてしまう。
# 元の spiece.model から語彙とスコアを読み、Unigram のモデルを組み立て直す
sp = spm.SentencePieceProcessor(model_file=hf_hub_download(name, "spiece.model"))
tokenizer.backend_tokenizer.model = models.Unigram(
    [(sp.id_to_piece(i), sp.get_score(i)) for i in range(sp.get_piece_size())], unk_id=sp.unk_id())
model = AutoModelForQuestionAnswering.from_pretrained(name).eval()

def answer(question, context):
    enc = tokenizer(question, context, return_tensors="pt", truncation="only_second", max_length=512, return_offsets_mapping=True)
    offsets = enc.pop("offset_mapping")[0]
    with torch.no_grad():
        out = model(**enc)
    is_context = torch.tensor([sid == 1 for sid in enc.sequence_ids(0)])          # 文書側のトークンだけを候補にする
    start = out.start_logits[0].masked_fill(~is_context, -1e9).log_softmax(-1)
    end = out.end_logits[0].masked_fill(~is_context, -1e9).log_softmax(-1)
    scores = start[:, None] + end[None, :]                                        # 全 (開始, 終了) の組のスコア
    scores = scores.triu().masked_fill(torch.ones_like(scores).tril(-1).bool(), -1e9)   # 開始 <= 終了
    scores = scores.masked_fill(torch.ones_like(scores).triu(30).bool(), -1e9)          # 長さ 30 トークンまで
    s, e = divmod(scores.argmax().item(), scores.shape[1])
    return context[offsets[s][0]:offsets[e][1]], round(scores[s, e].exp().item(), 3)

context = ("富士山は、静岡県と山梨県にまたがる活火山である。標高は 3776 メートルで、日本の最高峰である。"
           "2013 年に「富士山-信仰の対象と芸術の源泉」としてユネスコの世界文化遺産に登録された。")
for q in ["富士山の高さは？", "富士山はどこにありますか？", "世界遺産に登録されたのはいつ？", "富士山を最初に登った人は誰？"]:
    print(q, "→", answer(q, context))
```

- モデルは各トークンについて「答えの開始位置らしさ」「終了位置らしさ」を出力する。その和が最大になる区間を答えとする。
- 文書に答えがない質問（最後の例）にも、それらしい箇所を抜き出してしまう。スコアにしきい値を設けるか、回答不能も学習したモデルを使う。
- 生成型の LLM と違い、答えは必ず原文の一部なので、根拠の箇所が確実に分かる。
- transformers 5 系（5.0〜5.23 で確認）では、このモデルのトークナイザー（T5Tokenizer、SentencePiece の Unigram）が未知語の ID を 2 に固定して作られる。このモデルでは 2 は文末記号 `</s>` なので、語彙にない文字（大文字の英字など）が `</s>` として入力され、答えの位置がずれたり文の区切りと誤解されたりする。上のコードでは元の `spiece.model` から Unigram のモデルを組み立て直し、transformers 4 系と同じ分割に戻している（日本語と英語の 315 文で一致を確認）。上流で直れば不要になる。

## 質問応答を完全一致と F1 で評価する {qa_em_f1}

質問応答の出力を、正解と完全に一致したか (EM)、部分的にどれくらい重なっているか (F1) で評価したい。


```python
import re
import unicodedata
from collections import Counter
from fugashi import Tagger

tagger = Tagger("-Owakati")

def normalize(text):
    text = unicodedata.normalize("NFKC", text).lower()
    return re.sub(r"[\s。、．，,.!?！？「」『』()（）]", "", text)

def exact_match(pred, golds):
    return float(any(normalize(pred) == normalize(g) for g in golds))

def token_f1(pred, golds, unit="char"):
    """正解が複数あるときは、最も高い F1 を採用する"""
    tok = (lambda s: list(normalize(s))) if unit == "char" else (lambda s: tagger.parse(s).split())
    best = 0.0
    for g in golds:
        p, t = tok(pred), tok(g)
        common = sum((Counter(p) & Counter(t)).values())
        if common:
            prec, rec = common / len(p), common / len(t)
            best = max(best, 2 * prec * rec / (prec + rec))
    return best

examples = [("富士山", ["富士山"]), ("富士山です。", ["富士山"]), ("3776 メートル", ["3776m", "3776メートル"]),
            ("１９４５年", ["1945年", "1945"]), ("夏目漱石と森鴎外", ["夏目漱石"]), ("わかりません", ["東京"])]
for pred, golds in examples:
    print(f"EM {exact_match(pred, golds):.0f}  F1(文字) {token_f1(pred, golds):.2f}  F1(単語) {token_f1(pred, golds, 'word'):.2f}  {pred!r} / {golds}")
```

- 英語の SQuAD では、冠詞と句読点を除いて小文字化し、空白区切りの単語で F1 を計算する。日本語（JSQuAD など）では文字単位の F1 が使われる。
- LLM の出力は「富士山です。」のように文になりがちで、EM が不当に低くなる。プロンプトで形式を指定するか、「正解を含むか」での判定や LLM による判定を併用する。

## 文書に答えがない質問を「答えなし」と判定する {qa_unanswerable}

抽出型 QA は、文書に答えがなくてもどこかを抜き出してしまう。答えがあるかどうかを判定して、ないときは「答えなし」と返したい。

```python
import torch
from transformers import AutoModelForQuestionAnswering, AutoTokenizer

name = "deepset/xlm-roberta-base-squad2"   # 答えのない質問を含む SQuAD 2.0 で学習した多言語モデル
tokenizer = AutoTokenizer.from_pretrained(name)
model = AutoModelForQuestionAnswering.from_pretrained(name).eval()

@torch.no_grad()
def answer(question, context, max_len=30):
    enc = tokenizer(question, context, return_tensors="pt", truncation="only_second", max_length=512, return_offsets_mapping=True)
    offsets = enc.pop("offset_mapping")[0]
    out = model(**enc)
    start, end = out.start_logits[0], out.end_logits[0]
    null_score = (start[0] + end[0]).item()            # 先頭の特殊トークンを指す = 「答えなし」のスコア
    in_context = torch.tensor([s == 1 for s in enc.sequence_ids(0)])
    scores = start[:, None] + end[None, :]
    scores = scores.masked_fill(~in_context[:, None] | ~in_context[None, :], -1e9)
    scores = scores.triu().masked_fill(torch.ones_like(scores).tril(-1).bool(), -1e9).masked_fill(torch.ones_like(scores).triu(max_len).bool(), -1e9)
    s, e = divmod(scores.argmax().item(), scores.shape[1])
    return context[offsets[s][0]:offsets[e][1]], scores[s, e].item(), null_score

context = ("富士山は、静岡県と山梨県にまたがる活火山である。標高は 3776 メートルで、日本の最高峰である。"
           "2013 年に「富士山-信仰の対象と芸術の源泉」としてユネスコの世界文化遺産に登録された。")
questions = [("富士山の高さは？", True), ("富士山はどこにありますか？", True), ("世界遺産に登録されたのはいつ？", True),
             ("富士山に初めて登った人は誰ですか？", False), ("富士山の最後の噴火は何年ですか？", False), ("琵琶湖の面積は？", False)]
for q, answerable in questions:
    span, span_score, null_score = answer(q, context)
    diff = span_score - null_score                      # 差が小さい (または負) なら「答えなし」
    print(f"{'答えあり' if answerable else '答えなし'} | 差 {diff:6.1f} | {span[:24]!r:30} | {q}")
```

```python
def answer_or_none(question, context, threshold=0.0):
    span, span_score, null_score = answer(question, context)
    return span if span_score - null_score > threshold else None

for q, _ in questions:
    print(answer_or_none(q, context), "←", q)
```

- SQuAD 2.0 形式で学習したモデルは、答えがないとき、先頭の特殊トークンを開始・終了位置として指すように学習されている。「最良の区間のスコア − 答えなしのスコア」が判定の材料になる。
- しきい値は、答えのある質問とない質問の両方を含む検証データで決める。高くすると「答えなし」が増え（誤答が減り、見逃しが増える）、低くするとその逆。
- 答えなしを学習していないモデル（SQuAD 1.1 や JSQuAD で学習したもの）では、この差は当てにならない。その場合は区間の確率にしきい値を設けるしかなく、精度は低い。
- 多言語モデルを英語のデータで学習したものなので、日本語専用のモデルより抽出の精度は落ちる。日本語で答えなしを含むデータを用意できるなら、それで追加学習するのが望ましい。

```python
from datasets import load_dataset
from transformers import AutoModelForQuestionAnswering, AutoTokenizer

ds = load_dataset("sbintuitions/JSQuAD")
train = ds["train"].shuffle(seed=0).select(range(8000))      # 短時間で終わるように一部だけ使う (全体は約 6.3 万件)
valid = ds["validation"].select(range(1000))

model_name = "xlm-roberta-base"                               # 文字位置 (offset mapping) を返せるトークナイザを持つモデル
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForQuestionAnswering.from_pretrained(model_name)
MAX_LEN, STRIDE = 384, 128
```

```python
def prepare_train(batch):
    """長い文書は重なりを持たせて複数の窓に分け、各窓について答えの開始・終了トークン位置を求める"""
    enc = tokenizer(batch["question"], batch["context"], truncation="only_second", max_length=MAX_LEN, stride=STRIDE,
                    return_overflowing_tokens=True, return_offsets_mapping=True, padding="max_length")
    starts, ends = [], []
    for i, offsets in enumerate(enc["offset_mapping"]):
        answers = batch["answers"][enc["overflow_to_sample_mapping"][i]]
        a_start = answers["answer_start"][0]
        a_end = a_start + len(answers["text"][0])
        seq = enc.sequence_ids(i)
        ctx_start, ctx_end = seq.index(1), len(seq) - 1 - seq[::-1].index(1)
        if offsets[ctx_start][0] > a_start or offsets[ctx_end][1] < a_end:      # この窓に答えが入っていない
            starts.append(0); ends.append(0)
        else:
            s = ctx_start
            while s <= ctx_end and offsets[s][0] <= a_start:
                s += 1
            e = ctx_end
            while e >= ctx_start and offsets[e][1] >= a_end:
                e -= 1
            starts.append(s - 1); ends.append(e + 1)
    enc["start_positions"], enc["end_positions"] = starts, ends
    enc.pop("offset_mapping"); enc.pop("overflow_to_sample_mapping")
    return enc

def prepare_valid(batch):
    enc = tokenizer(batch["question"], batch["context"], truncation="only_second", max_length=MAX_LEN, stride=STRIDE,
                    return_overflowing_tokens=True, return_offsets_mapping=True, padding="max_length")
    enc["example_id"] = [batch["id"][j] for j in enc["overflow_to_sample_mapping"]]
    enc["offset_mapping"] = [[o if seq == 1 else None for o, seq in zip(offsets, enc.sequence_ids(i))]     # 文書以外の位置は None にしておく
                             for i, offsets in enumerate(enc["offset_mapping"])]
    enc.pop("overflow_to_sample_mapping")
    return enc

train_enc = train.map(prepare_train, batched=True, remove_columns=train.column_names)
valid_enc = valid.map(prepare_valid, batched=True, remove_columns=valid.column_names)
print(len(train), "→", len(train_enc), "窓 /", len(valid), "→", len(valid_enc), "窓")
```

```python
import collections
import re
import unicodedata
import numpy as np

def postprocess(examples, features, start_logits, end_logits, n_best=20, max_answer_len=30):
    """窓ごとの予測を質問単位にまとめ、スコアが最大の区間を答えとする"""
    by_example = collections.defaultdict(list)
    for i, f in enumerate(features):
        by_example[f["example_id"]].append(i)
    predictions = {}
    for ex in examples:
        best = (-1e9, "")
        for i in by_example[ex["id"]]:
            offsets = features[i]["offset_mapping"]
            for s in np.argsort(start_logits[i])[::-1][:n_best]:
                for e in np.argsort(end_logits[i])[::-1][:n_best]:
                    if offsets[s] is None or offsets[e] is None or e < s or e - s + 1 > max_answer_len:
                        continue
                    score = start_logits[i][s] + end_logits[i][e]
                    if score > best[0]:
                        best = (score, ex["context"][offsets[s][0]:offsets[e][1]])
        predictions[ex["id"]] = best[1]
    return predictions

norm = lambda s: re.sub(r"[\s。、．，,.!?！？「」『』()（）]", "", unicodedata.normalize("NFKC", s).lower())

def char_f1(pred, gold):
    p, g = list(norm(pred)), list(norm(gold))
    common = sum((collections.Counter(p) & collections.Counter(g)).values())
    return 2 * common / (len(p) + len(g)) if common else 0.0

out = trainer.predict(valid_enc.remove_columns(["example_id", "offset_mapping"]))
preds = postprocess(valid, valid_enc, *out.predictions)
em = np.mean([any(norm(preds[ex["id"]]) == norm(a) for a in ex["answers"]["text"]) for ex in valid])
f1 = np.mean([max(char_f1(preds[ex["id"]], a) for a in ex["answers"]["text"]) for ex in valid])
print(f"EM = {em:.3f}  F1 (文字) = {f1:.3f}")
for ex in valid.select(range(3)):
    print(ex["question"], "→", preds[ex["id"]], "/ 正解:", ex["answers"]["text"][0])
```

- 抽出型 QA は「答えの開始位置」と「終了位置」をそれぞれトークン分類として学習する。前処理の中心は、文字位置で与えられた答えをトークン位置に直すこと。
- 長い文書は、重なり (`stride`) を持たせた複数の窓に分ける。答えが入っていない窓は、先頭の特殊トークンを正解にする。
- 予測は窓単位で出るので、同じ質問の窓をまとめて最良の区間を選ぶ後処理が要る。開始 > 終了や長すぎる区間はここで除く。
- T4 GPU で 8000 件・2 エポックが 10 分程度。この 8000 件では EM が 0.75 前後、文字 F1 が 0.88 前後。全データ（約 6.3 万件）で学習すると EM は 0.82、文字 F1 は 0.94 程度まで上がるが、学習には 5 倍以上の時間がかかる。日本語専用の BERT のほうが精度は高いが、文字位置を返さないトークナイザだと位置合わせを自分で実装する必要がある。

```python
import numpy as np
from datasets import load_dataset
from sentence_transformers import SentenceTransformer

ds = load_dataset("sbintuitions/JCommonsenseQA", split="validation").select(range(100))   # 常識を問う 5 択問題 (JGLUE)
ex = ds[0]
print(ex["question"], [ex[f"choice{i}"] for i in range(5)], "正解:", ex["label"])

# 1. 埋め込み: 質問と最も似ている選択肢を選ぶ
embedder = SentenceTransformer("intfloat/multilingual-e5-small")
q_emb = embedder.encode(["query: " + e["question"] for e in ds], normalize_embeddings=True)
c_emb = embedder.encode(["query: " + e[f"choice{i}"] for e in ds for i in range(5)], normalize_embeddings=True).reshape(len(ds), 5, -1)
pred_emb = np.einsum("nd,ncd->nc", q_emb, c_emb).argmax(axis=1)
gold = np.array(ds["label"])
print(f"偶然の正解率 0.200 / 埋め込みの類似度 {(pred_emb == gold).mean():.3f}")
```

```python
import sentencepiece as spm
import torch
from huggingface_hub import hf_hub_download
from tokenizers import models
from transformers import AutoModelForCausalLM, AutoTokenizer

name = "sbintuitions/sarashina2.2-3b-instruct-v0.1"   # 3B の日本語モデル (T4 GPU で 1 問あたり 1 秒未満)
tokenizer = AutoTokenizer.from_pretrained(name)
# transformers 5 系は、このモデルの SentencePiece (Unigram) を BPE に変換してしまい、学習時と違う分割になる。
# 元の tokenizer.model から語彙とスコアを読み、分割の方式だけを Unigram に戻す (特殊トークンやチャットテンプレートはそのまま)
sp = spm.SentencePieceProcessor(model_file=hf_hub_download(name, "tokenizer.model"))
tokenizer.backend_tokenizer.model = models.Unigram(
    [(sp.id_to_piece(i), sp.get_score(i)) for i in range(sp.get_piece_size())], unk_id=sp.unk_id(), byte_fallback=True)
BF16 = torch.cuda.get_device_capability()[0] >= 8      # bfloat16 は Ampere (A100, L4) 以降。T4 では float16 を使う
llm = AutoModelForCausalLM.from_pretrained(name, dtype=torch.bfloat16 if BF16 else torch.float16, device_map="auto").eval()
digit_ids = [tokenizer(str(i), add_special_tokens=False)["input_ids"][-1] for i in range(1, 6)]   # 「1」〜「5」のトークン

@torch.no_grad()
def choose(question, choices):
    """選択肢に番号を振り、次のトークンとして各番号が出る確率を比べる (生成はしないので 1 回の計算で済む)"""
    listing = "\n".join(f"{i}. {c}" for i, c in enumerate(choices, 1))
    prompt = f"次の質問の答えとして最も適切なものを選び、番号だけを答えてください。\n\n質問: {question}\n{listing}"
    text = tokenizer.apply_chat_template([{"role": "user", "content": prompt}], add_generation_prompt=True, tokenize=False)
    ids = tokenizer(text, add_special_tokens=False, return_tensors="pt")["input_ids"].to(llm.device)
    probs = llm(ids).logits[0, -1, digit_ids].float().softmax(-1)
    return int(probs.argmax()), probs.max().item()

results = [choose(e["question"], [e[f"choice{i}"] for i in range(5)]) for e in ds]   # T4 なら 100 問で 1 分程度
pred_llm, conf = np.array([r[0] for r in results]), np.array([r[1] for r in results])
print(f"LLM に番号を選ばせる {(pred_llm == gold).mean():.3f}")
```

```python
# 確信度の高い問題ほど正解しているか / 間違えた問題を見る
order = np.argsort(-conf)
print(f"確信度の上位 50 問の正解率 {(pred_llm == gold)[order[:50]].mean():.3f} / 下位 50 問 {(pred_llm == gold)[order[50:]].mean():.3f}")
for i in np.where(pred_llm != gold)[0][:3]:
    e = ds[int(i)]
    print(f"{e['question']} → 予測: {e[f'choice{pred_llm[i]}']} / 正解: {e[f'choice{gold[i]}']}")
```

- 選択肢に番号を振って、番号のトークンの確率を比べると、選択肢が長くても公平に比較でき、1 問につき 1 回の計算で済む。
- LLM は選択肢の位置に偏りを持つことがある（1 番を選びやすいなど）。選択肢の順序を入れ替えて複数回解かせ、多数決をとると安定する。
- 埋め込みの類似度は「質問と似た語」を選ぶだけなので、常識や推論が要る問題には弱い。ただし、偶然よりどれだけ上かを知るためのベースラインとして有用。
- 文書を読んで選ぶ形式（読解問題）なら、プロンプトに文書を加える。知識が足りない問題は、検索した文書を付ける (RAG) と改善する。

## 表の内容について質問に答える {table_qa}

表（スプレッドシートや DataFrame）の内容について、自然言語の質問に答えたい。「合計は？」「一番大きいのは？」のような集計も扱いたい。

```python
import pandas as pd
import torch
from transformers import TapasForQuestionAnswering, TapasTokenizer

name = "google/tapas-base-finetuned-wtq"    # 表に対する質問応答のモデル (英語)。セルの選択と集計の種類を予測する
tapas_tok = TapasTokenizer.from_pretrained(name)
tapas = TapasForQuestionAnswering.from_pretrained(name).eval()

table = pd.DataFrame({"City": ["Tokyo", "Osaka", "Nagoya", "Sapporo"], "Prefecture": ["Tokyo", "Osaka", "Aichi", "Hokkaido"],
                      "Population": ["13960000", "2750000", "2330000", "1970000"], "Area": ["2194", "225", "326", "1121"]})   # TAPAS にはすべて文字列で渡す
queries = ["Which city has the largest population?", "What is the total population of Osaka and Nagoya?",
           "How many cities have an area larger than 300?", "What is the average area?", "Which prefecture is Nagoya in?"]

enc = tapas_tok(table=table, queries=queries, padding="max_length", return_tensors="pt")
with torch.no_grad():
    out = tapas(**enc)
coords, aggs = tapas_tok.convert_logits_to_predictions(enc, out.logits, out.logits_aggregation)
AGG = {0: "NONE", 1: "SUM", 2: "AVERAGE", 3: "COUNT"}
for q, cells, agg in zip(queries, coords, aggs):
    values = [table.iat[c] for c in cells]
    nums = [float(v) for v in values if v.replace(".", "").isdigit()]
    result = {"NONE": ", ".join(values), "SUM": sum(nums), "AVERAGE": sum(nums) / max(1, len(nums)), "COUNT": len(values)}[AGG[agg]]
    print(f"{q}\n   {AGG[agg]} {values} → {result}")
```

```python
# 日本語の表: 表を Markdown にして LLM に読ませる
import sentencepiece as spm
from huggingface_hub import hf_hub_download
from tokenizers import models
from transformers import AutoModelForCausalLM, AutoTokenizer

name = "sbintuitions/sarashina2.2-3b-instruct-v0.1"
tokenizer = AutoTokenizer.from_pretrained(name)
# transformers 5 系は、このモデルの SentencePiece (Unigram) を BPE に変換してしまい、学習時と違う分割になる。
# 元の tokenizer.model から語彙とスコアを読み、分割の方式だけを Unigram に戻す (特殊トークンやチャットテンプレートはそのまま)
sp = spm.SentencePieceProcessor(model_file=hf_hub_download(name, "tokenizer.model"))
tokenizer.backend_tokenizer.model = models.Unigram(
    [(sp.id_to_piece(i), sp.get_score(i)) for i in range(sp.get_piece_size())], unk_id=sp.unk_id(), byte_fallback=True)
llm = AutoModelForCausalLM.from_pretrained(name).eval()

df = pd.DataFrame({"都市": ["東京", "大阪", "名古屋", "札幌"], "都道府県": ["東京都", "大阪府", "愛知県", "北海道"],
                   "人口 (万人)": [1396, 275, 233, 197], "面積 (km2)": [2194, 225, 326, 1121]})
markdown = "| " + " | ".join(df.columns) + " |\n|" + "---|" * len(df.columns) + "\n" + "\n".join("| " + " | ".join(map(str, r)) + " |" for r in df.values)

def ask(question):
    prompt = f"次の表だけを見て、質問に短く答えてください。\n\n{markdown}\n\n質問: {question}"
    inputs = tokenizer.apply_chat_template([{"role": "user", "content": prompt}], add_generation_prompt=True, return_tensors="pt", return_dict=True)
    with torch.no_grad():
        out = llm.generate(**inputs, max_new_tokens=60, do_sample=False, pad_token_id=tokenizer.eos_token_id)
    return tokenizer.decode(out[0, inputs["input_ids"].shape[1]:], skip_special_tokens=True).strip()

checks = {"人口が一番多い都市はどこですか？": df.loc[df["人口 (万人)"].idxmax(), "都市"], "名古屋は何県にありますか？": "愛知県",
          "面積が 300 km2 より大きい都市はいくつありますか？": str((df["面積 (km2)"] > 300).sum()),
          "大阪と名古屋の人口の合計は何万人ですか？": str(275 + 233), "4 都市の面積の平均は？": str(df["面積 (km2)"].mean())}
for q, truth in checks.items():
    print(f"{q}\n   LLM: {ask(q)}\n   pandas で計算した正解: {truth}")
```

- TAPAS は、答えに使うセルと集計の種類（なし・合計・平均・個数）を予測する。計算そのものはプログラムが行うので、選んだセルが正しければ数値は必ず合う。
- LLM に表を読ませる方法は手軽で日本語でも使えるが、セルの参照は得意でも、合計や平均などの計算は間違えやすい。計算が要る質問は、上のように pandas の結果と突き合わせて確かめる。
- 行数の多い表はプロンプトに入りきらない。質問から pandas のコードや SQL を生成させて実行する方法 (text-to-SQL) が、大きな表では現実的。
- 表を文字列にする形式（Markdown、CSV、HTML）で精度が変わる。列名と単位を明記するのが基本。

```python
import re
import sentencepiece as spm
import torch
from fugashi import Tagger
from huggingface_hub import hf_hub_download
from tokenizers import models
from transformers import AutoModelForCausalLM, AutoModelForQuestionAnswering, AutoTokenizer

name = "sbintuitions/sarashina2.2-3b-instruct-v0.1"
tokenizer = AutoTokenizer.from_pretrained(name)
# transformers 5 系は、このモデルの SentencePiece (Unigram) を BPE に変換してしまい、学習時と違う分割になる。
# 元の tokenizer.model から語彙とスコアを読み、分割の方式だけを Unigram に戻す (特殊トークンやチャットテンプレートはそのまま)
sp = spm.SentencePieceProcessor(model_file=hf_hub_download(name, "tokenizer.model"))
tokenizer.backend_tokenizer.model = models.Unigram(
    [(sp.id_to_piece(i), sp.get_score(i)) for i in range(sp.get_piece_size())], unk_id=sp.unk_id(), byte_fallback=True)
llm = AutoModelForCausalLM.from_pretrained(name).eval()

def chat(messages, max_new_tokens=60):
    inputs = tokenizer.apply_chat_template(messages, add_generation_prompt=True, return_tensors="pt", return_dict=True)
    with torch.no_grad():
        out = llm.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False, pad_token_id=tokenizer.eos_token_id)
    return tokenizer.decode(out[0, inputs["input_ids"].shape[1]:], skip_special_tokens=True).strip()

passage = ("琵琶湖は、滋賀県にある日本最大の湖である。面積は約 670 平方キロメートルで、滋賀県の面積のおよそ 6 分の 1 を占める。"
           "約 400 万年前に誕生したとされ、世界でも有数の古代湖である。湖から流れ出る川は瀬田川だけで、下流では淀川と名前を変えて大阪湾に注ぐ。"
           "近畿地方の約 1450 万人の生活用水を支えており、「近畿の水がめ」と呼ばれる。")

# 1. 答えの候補を選ぶ: 数値を含む表現と固有名詞を、文書から機械的に取り出す
tagger = Tagger()
candidates = re.findall(r"約?\s*\d[\d,.]*\s*(?:万|億)?\s*(?:平方キロメートル|キロメートル|万年前|年前|分の\s*\d+|万人|人|年)", passage)
buf = []
for w in list(tagger(passage)) + [None]:
    is_proper = w is not None and w.feature.pos1 == "名詞" and w.feature.pos2 == "固有名詞"
    is_tail = w is not None and buf and w.feature.pos1 in ("名詞", "接尾辞") and w.feature.pos2 != "数詞"   # 琵琶 + 湖、滋賀 + 県 のように後ろに続く語
    if is_proper or is_tail:
        buf.append(w.surface)
    else:
        if buf:
            candidates.append("".join(buf))
        buf = []
candidates = list(dict.fromkeys(c.strip() for c in candidates))
print(candidates)
```

```python
# 2. 答えを指定して、それが答えになる質問を作らせる (例を 1 つ見せる)
INSTRUCTION = "次の文章を読めば答えられる質問を 1 つ作ってください。質問の答えは、指定された「答え」になるようにしてください。質問文だけを出力してください。"
SHOT = [{"role": "user", "content": f"{INSTRUCTION}\n\n# 文章\n富士山は、静岡県と山梨県にまたがる活火山である。標高は 3776 メートルで、日本の最高峰である。\n\n# 答え\n3776 メートル"},
        {"role": "assistant", "content": "富士山の標高は何メートルですか？"}]

def generate_question(passage, answer):
    return chat(SHOT + [{"role": "user", "content": f"{INSTRUCTION}\n\n# 文章\n{passage}\n\n# 答え\n{answer}"}]).split("\n")[0]

pairs = [(generate_question(passage, a), a) for a in candidates[:8]]
for q, a in pairs:
    print(f"Q: {q}\n   A: {a}")
```

```python
# 3. 往復検査: 作った質問を QA モデルに解かせて、指定した答えが返ってくるものだけを残す
qa_name = "tsmatz/roberta_qa_japanese"
qa_tok = AutoTokenizer.from_pretrained(qa_name)
# transformers 5 系の T5Tokenizer は未知語を文末記号 (</s>) の ID にしてしまうので、Unigram のモデルを組み立て直す (この章の「抽出型 QA」を参照)
sp = spm.SentencePieceProcessor(model_file=hf_hub_download(qa_name, "spiece.model"))
qa_tok.backend_tokenizer.model = models.Unigram([(sp.id_to_piece(i), sp.get_score(i)) for i in range(sp.get_piece_size())], unk_id=sp.unk_id())
qa = AutoModelForQuestionAnswering.from_pretrained(qa_name).eval()

@torch.no_grad()
def extract(question, context):
    enc = qa_tok(question, context, return_tensors="pt", truncation="only_second", max_length=512, return_offsets_mapping=True)
    offsets = enc.pop("offset_mapping")[0]
    out = qa(**enc)
    in_context = torch.tensor([s == 1 for s in enc.sequence_ids(0)])
    scores = out.start_logits[0][:, None] + out.end_logits[0][None, :]
    scores = scores.masked_fill(~in_context[:, None] | ~in_context[None, :], -1e9)
    scores = scores.triu().masked_fill(torch.ones_like(scores).tril(-1).bool(), -1e9).masked_fill(torch.ones_like(scores).triu(30).bool(), -1e9)
    s, e = divmod(scores.argmax().item(), scores.shape[1])
    return context[offsets[s][0]:offsets[e][1]]

squash = lambda s: re.sub(r"\s|約", "", s)
kept = []
for q, a in pairs:
    predicted = extract(q, passage)
    ok = squash(a) in squash(predicted) or squash(predicted) in squash(a)
    kept += [(q, a)] if ok and predicted else []
    print(f"{'○' if ok else '×'} {q} → QA モデルの答え: {predicted!r} (指定: {a!r})")
print(len(kept), "/", len(pairs), "組を採用")
```

- 「答えを先に決めてから質問を作らせる」と、答えが文書中にあることが保証され、抽出型 QA のデータとしてそのまま使える。
- 生成した質問には、答えが一意に決まらないもの、文書を読まなくても答えられるもの、指定した答えと対応しないものが混ざる。QA モデルで解き直して答えが一致するものだけを残す（往復検査）と、質がかなり揃う。
- 上の往復検査は「一方が他方を含む」で判定しているので、「日本最大」を答えに指定した「日本で何番目に大きい湖ですか？」のように、QA モデルが「日本」とだけ返した組も残ってしまう。厳しくするなら、文字単位の F1 にしきい値を設ける。
- RAG や検索の評価セットを作るときにも同じ方法が使える。ただし、文書の言い回しをなぞった質問になりやすく、実際のユーザーの質問より易しい点に注意する。
- 練習問題として使うなら、答えの候補の選び方が出題の質を決める。重要な概念を問うには、候補を人が選ぶか、重要語の抽出と組み合わせる。
